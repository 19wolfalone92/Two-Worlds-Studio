cmake_minimum_required(VERSION 3.22)
project(wolf_embedded LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_ALL_WARNINGS OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(LLAMA_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../third_party/llama.cpp")
if (NOT EXISTS "${LLAMA_ROOT}/include/llama.h")
    message(FATAL_ERROR "Pin ggml-org/llama.cpp into ${LLAMA_ROOT} before build")
endif()
add_subdirectory("${LLAMA_ROOT}" "${CMAKE_BINARY_DIR}/llama-build" EXCLUDE_FROM_ALL)
add_library(wolf_native SHARED wolf_native.cpp)
target_include_directories(wolf_native PRIVATE "${LLAMA_ROOT}/include")
target_link_libraries(wolf_native PRIVATE llama android log)
target_compile_options(wolf_native PRIVATE -O2 -fvisibility=hidden)
